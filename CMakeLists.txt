cmake_minimum_required(VERSION 3.25)
project(HibernationAgent VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(ALIYAT_ENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../AliyatEngine" CACHE PATH "AliyatEngine checkout")
if(MSVC)
    add_compile_options(/utf-8)
    include("${ALIYAT_ENGINE_ROOT}/cmake/AYMsvcNinjaDeps.cmake")
endif()
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(AY_BUILD_RUNTIME OFF CACHE BOOL "" FORCE)
set(AY_BUILD_APPLICATION OFF CACHE BOOL "" FORCE)
set(AY_BUILD_EDITOR OFF CACHE BOOL "" FORCE)
set(AY_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(AY_ENABLE_NETWORK OFF CACHE BOOL "" FORCE)
set(AY_ENABLE_SCRIPT OFF CACHE BOOL "" FORCE)
set(AY_ENABLE_HTTP ON CACHE BOOL "" FORCE)
set(AY_ENABLE_INFERENCE ON CACHE BOOL "" FORCE)
set(AY_BUILD_APPLICATION_SERVICES ON CACHE BOOL "" FORCE)
set(AY_ENABLE_EVENT_JOURNAL ON CACHE BOOL "" FORCE)
option(HA_BUILD_PLUGINS "Build the optional desktop/media/weather/speech plugins" ON)
set(HA_FFMPEG_SDK "${CMAKE_CURRENT_SOURCE_DIR}/out/deps/ffmpeg-sdk" CACHE PATH "Verified FFmpeg headers and existing engine binary import libraries")
if(HA_BUILD_PLUGINS)
    if(NOT EXISTS "${HA_FFMPEG_SDK}/lib/avcodec-62.lib")
        message(FATAL_ERROR "Run scripts/PrepareVideo.ps1, or configure HA_BUILD_PLUGINS=OFF for a core-only build")
    endif()
    add_library(HibernationFFmpeg INTERFACE)
    target_include_directories(HibernationFFmpeg INTERFACE "${HA_FFMPEG_SDK}/ffmpeg-8.1.2")
    foreach(_av avcodec-62 avformat-62 avutil-60 swresample-6)
        target_link_libraries(HibernationFFmpeg INTERFACE "${HA_FFMPEG_SDK}/lib/${_av}.lib")
    endforeach()
    set(AY_VIDEO_DECODE_FFMPEG_TARGET HibernationFFmpeg)
    file(GLOB AY_VIDEO_DECODE_RUNTIME_FILES "${HA_FFMPEG_SDK}/bin/*.dll")
    set(AY_ENABLE_VIDEO_DECODE ON CACHE BOOL "" FORCE)
else()
    set(AY_ENABLE_VIDEO_DECODE OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${ALIYAT_ENGINE_ROOT}" engine EXCLUDE_FROM_ALL)
# Explicitly compose the light plugin substrate; the full game Runtime remains off.
add_subdirectory("${ALIYAT_ENGINE_ROOT}/AYRuntime/AYCore" engine-role-core EXCLUDE_FROM_ALL)
add_subdirectory("${ALIYAT_ENGINE_ROOT}/AYRuntime/AYPlugin" engine-role-plugin EXCLUDE_FROM_ALL)
add_subdirectory("${ALIYAT_ENGINE_ROOT}/AYRuntime/AYAudio/pan" engine-audio-pan EXCLUDE_FROM_ALL)
if(HA_BUILD_PLUGINS)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/out/deps/miniaudio/miniaudio.h")
        set(MINIAUDIO_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/out/deps/miniaudio")
    endif()
    add_subdirectory("${ALIYAT_ENGINE_ROOT}/AYRuntime/AYAudio/playback" engine-audio-playback EXCLUDE_FROM_ALL)
    add_subdirectory("${ALIYAT_ENGINE_ROOT}/AYRuntime/AYVideo/playback" engine-video-playback EXCLUDE_FROM_ALL)
endif()
if(HA_BUILD_PLUGINS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/out/deps/webview2/sdk/build/native/include/WebView2.h")
    set(AY_WEBVIEW2_SDK "${CMAKE_CURRENT_SOURCE_DIR}/out/deps/webview2/sdk")
    add_subdirectory("${ALIYAT_ENGINE_ROOT}/AYFoundation/AYPlatform/webview" engine-webview EXCLUDE_FROM_ALL)
endif()
find_package(nlohmann_json CONFIG REQUIRED)
add_library(HibernationCore STATIC src/Application.cpp src/Policy.cpp src/Vault.cpp src/TurnProtocol.cpp src/RoleState.cpp src/PluginRuntime.cpp src/Review.cpp src/DesktopServices.cpp src/ImageEncoding.cpp src/ReactionCache.cpp)
target_include_directories(HibernationCore PUBLIC include)
target_link_libraries(HibernationCore PUBLIC AYApplicationServices AYInference AYNetworkHttp AYIO AYEventJournal AYCrypto AYPlugin nlohmann_json::nlohmann_json crypt32)
target_link_libraries(HibernationCore PUBLIC AYPlatform user32 gdi32 ole32 windowscodecs)
if(TARGET AYPlatformWebView)
    target_link_libraries(HibernationCore PRIVATE AYPlatformWebView)
    target_compile_definitions(HibernationCore PRIVATE HA_HAS_WEBVIEW=1)
endif()
target_compile_definitions(HibernationCore PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(hibernation-agent src/Main.cpp src/DesktopHost.cpp src/ChatBubbles.cpp src/StartupSplash.cpp)
target_link_libraries(hibernation-agent PRIVATE HibernationCore user32 gdi32 shell32 comctl32 ole32)
target_compile_definitions(hibernation-agent PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(HA_BUILD_PLUGINS)
    file(GLOB _ha_media_runtime "${HA_FFMPEG_SDK}/bin/*.dll")
    add_custom_command(TARGET hibernation-agent POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${_ha_media_runtime} "$<TARGET_FILE_DIR:hibernation-agent>" COMMAND_EXPAND_LISTS)
    add_subdirectory(plugins)
    add_dependencies(hibernation-agent HibernationPlugins)
endif()
if(MSVC)
    target_compile_options(HibernationCore PRIVATE /utf-8 /W4)
    target_compile_options(hibernation-agent PRIVATE /utf-8 /W4)
endif()
if(BUILD_TESTING)
    add_executable(Hibernation_AppearancePreview EXCLUDE_FROM_ALL tests/AppearancePreview.cpp)
    target_include_directories(Hibernation_AppearancePreview PRIVATE include)
    target_compile_definitions(Hibernation_AppearancePreview PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(Hibernation_AppearancePreview PRIVATE user32 gdi32 gdiplus)
    add_executable(Hibernation_UnitTests tests/Unit.cpp tests/VaultCompatibility.cpp)
    add_executable(Hibernation_WorldArchiveFixture EXCLUDE_FROM_ALL tests/WorldArchiveFixture.cpp)
    target_link_libraries(Hibernation_WorldArchiveFixture PRIVATE HibernationCore)
    target_link_libraries(Hibernation_UnitTests PRIVATE HibernationCore AYTest)
    target_compile_definitions(Hibernation_UnitTests PRIVATE HA_PROJECT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
    if(HA_BUILD_PLUGINS)
        target_compile_definitions(Hibernation_UnitTests PRIVATE HA_TEST_PLUGIN_ROOT="${CMAKE_BINARY_DIR}/plugins")
        add_dependencies(Hibernation_UnitTests HibernationPlugins)
    endif()
    ay_add_test(Hibernation_UnitTests CASE_REPORT)
    find_program(HA_PWSH pwsh REQUIRED)
    find_package(httplib CONFIG REQUIRED)
    add_executable(Hibernation_TestProvider tests/TestProvider.cpp)
    target_link_libraries(Hibernation_TestProvider PRIVATE httplib::httplib nlohmann_json::nlohmann_json)
    if(HA_BUILD_PLUGINS)
        target_sources(Hibernation_TestProvider PRIVATE "${ALIYAT_ENGINE_ROOT}/AYRuntime/AYVideo/unittest/FFmpegTestMedia.cpp")
        target_include_directories(Hibernation_TestProvider PRIVATE "${ALIYAT_ENGINE_ROOT}/AYRuntime/AYVideo/unittest")
        target_link_libraries(Hibernation_TestProvider PRIVATE HibernationFFmpeg AYTest)
        target_compile_definitions(Hibernation_TestProvider PRIVATE HA_TEST_MEDIA=1)
        add_test(NAME Hibernation_ProtocolIntegration COMMAND "${HA_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/ProtocolIntegration.ps1" -Executable "$<TARGET_FILE:hibernation-agent>" -ProviderExecutable "$<TARGET_FILE:Hibernation_TestProvider>" -ProjectRoot "${CMAKE_CURRENT_SOURCE_DIR}" -ScratchRoot "${CMAKE_BINARY_DIR}/integration")
        set_tests_properties(Hibernation_ProtocolIntegration PROPERTIES TIMEOUT 180)
    endif()
    add_test(NAME Hibernation_ProcessIntegration COMMAND "${HA_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Integration.ps1" -Executable "$<TARGET_FILE:hibernation-agent>" -ProviderExecutable "$<TARGET_FILE:Hibernation_TestProvider>" -ProjectRoot "${CMAKE_CURRENT_SOURCE_DIR}" -ScratchRoot "${CMAKE_BINARY_DIR}/integration")
    set_tests_properties(Hibernation_ProcessIntegration PROPERTIES TIMEOUT 120)
    add_test(NAME Hibernation_ContinuityIntegration COMMAND "${HA_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Continuity.ps1" -Executable "$<TARGET_FILE:hibernation-agent>" -ProviderExecutable "$<TARGET_FILE:Hibernation_TestProvider>" -ProjectRoot "${CMAKE_CURRENT_SOURCE_DIR}" -ScratchRoot "${CMAKE_BINARY_DIR}/integration")
    set_tests_properties(Hibernation_ContinuityIntegration PROPERTIES TIMEOUT 180)
    add_test(NAME Hibernation_ForegroundPriority COMMAND "${HA_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/ForegroundPriority.ps1" -Executable "$<TARGET_FILE:hibernation-agent>" -ProviderExecutable "$<TARGET_FILE:Hibernation_TestProvider>" -ProjectRoot "${CMAKE_CURRENT_SOURCE_DIR}" -ScratchRoot "${CMAKE_BINARY_DIR}/integration")
    set_tests_properties(Hibernation_ForegroundPriority PROPERTIES TIMEOUT 90)
    add_test(NAME Hibernation_LauncherCheck COMMAND "${HA_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Launcher.ps1" -ProjectRoot "${CMAKE_CURRENT_SOURCE_DIR}" -BuildRoot "${CMAKE_BINARY_DIR}")
    set_tests_properties(Hibernation_LauncherCheck PROPERTIES TIMEOUT 60)
    add_custom_target(Hibernation_Checks DEPENDS Hibernation_UnitTests Hibernation_TestProvider AYApplication_ServiceTests AYApplication_ServiceIntegration AYStorage_Test AYInference_Tests AYPluginTest AYCrypto_Test AYNetwork_HttpTests)
    add_dependencies(Hibernation_Checks AYAudio_PanTests)
    add_dependencies(Hibernation_Checks Hibernation_WorldArchiveFixture)
    if(HA_BUILD_PLUGINS)
        add_dependencies(Hibernation_Checks AYVideoPlayback_Tests AYAudioPlayback_Tests AYAudioOutput_Tests)
    endif()
    add_dependencies(Hibernation_Checks AYPlatform_Test)
    if(TARGET AYPlatform_WebViewTests)
        add_dependencies(Hibernation_Checks AYPlatform_WebViewTests)
    endif()
    if(HA_BUILD_PLUGINS)
        add_dependencies(Hibernation_Checks AYVideoDecode_Test)
    endif()
endif()
