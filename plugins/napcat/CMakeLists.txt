find_package(httplib CONFIG REQUIRED)
add_library(Hibernation_Napcat SHARED Napcat.cpp)
target_include_directories(Hibernation_Napcat PRIVATE "${PROJECT_SOURCE_DIR}/include")
target_compile_definitions(Hibernation_Napcat PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(Hibernation_Napcat PRIVATE HibernationCore httplib::httplib bcrypt)
set_target_properties(Hibernation_Napcat PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/napcat")
configure_file(plugin.json "${CMAKE_BINARY_DIR}/plugins/napcat/plugin.json" COPYONLY)
add_dependencies(HibernationPlugins Hibernation_Napcat)
if(BUILD_TESTING)
    add_executable(Hibernation_NapcatImageProbe EXCLUDE_FROM_ALL ImageProbe.cpp)
    target_link_libraries(Hibernation_NapcatImageProbe PRIVATE HibernationCore bcrypt)
    target_compile_definitions(Hibernation_NapcatImageProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_executable(Hibernation_NapcatTests Tests.cpp)
    target_link_libraries(Hibernation_NapcatTests PRIVATE HibernationCore httplib::httplib bcrypt)
    target_compile_definitions(Hibernation_NapcatTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN HA_PROJECT_ROOT="${PROJECT_SOURCE_DIR}" HA_PLUGIN_ROOT="${CMAKE_BINARY_DIR}/plugins")
    add_dependencies(Hibernation_NapcatTests Hibernation_Napcat)
    add_test(NAME Hibernation_NapcatTests COMMAND Hibernation_NapcatTests "${CMAKE_BINARY_DIR}/napcat-tests")
    set_tests_properties(Hibernation_NapcatTests PROPERTIES TIMEOUT 120)
    add_test(NAME Hibernation_ChannelStyle COMMAND Hibernation_NapcatTests "${CMAKE_BINARY_DIR}/channel-style-tests" style)
    set_tests_properties(Hibernation_ChannelStyle PROPERTIES TIMEOUT 60)
endif()
